  const max=Math.max(income,expense,cash,due,1);
  const pct=v=>Math.max(4,Math.round(v/max*100));
  const p=Number(x.attendanceToday.present)||0, a=Number(x.attendanceToday.absent)||0, total=p+a, pp=total?Math.round(p/total*100):0;
  const periods=['দৈনিক','সাপ্তাহিক','মাসিক','বাৎসরিক'];
  $('dashboardCharts').innerHTML=`
    <div class="chart-card">
      <div class="chart-title">📊 আয় + ব্যয় + ক্যাশ + বকেয়া — সময়ভিত্তিক সামারি</div>
      <div class="period-tabs">${periods.map((v,i)=>`<button class="period-tab ${i===0?'active':''}" onclick="selectDashboardPeriod(this,'${v}')">${v}</button>`).join('')}</div>
      <div class="bar-chart">
        ${[['আয়',income,'income'],['ব্যয়',expense,'expense'],['ক্যাশ',cash,'cash'],['বকেয়া',due,'expense']].map(v=>`<div class="bar-group"><div class="bar-pair"><div class="bar ${v[2]}" style="height:${pct(v[1])}%" title="${v[1]}"></div></div><div class="bar-label">${v[0]}</div></div>`).join('')}
      </div>
      <div class="legend"><span><i class="dot" style="background:#20a464"></i>আয়</span><span><i class="dot" style="background:#e33b4a"></i>ব্যয়/বকেয়া</span><span><i class="dot" style="background:#248bd0"></i>ক্যাশ</span></div>
    </div>
    <div class="chart-card">
      <div class="chart-title">📈 দৈনিক উপস্থিতি / অনুপস্থিতি</div>
      <div class="attendance-box">
        <div class="attendance-ring" style="--present:${pp}%"><span>${pp}%</span></div>
        <div class="attendance-counts"><div class="attendance-count"><b>${p}</b>উপস্থিত</div><div class="attendance-count"><b>${a}</b>অনুপস্থিত</div></div>
      </div>
    </div>`;
}

async function uploadMedia(){
  const input=$('mediaFiles');
  if(!input||!input.files||!input.files.length)return toast('প্রথমে ফটো/ভিডিও/অডিও নির্বাচন করুন।');
  let uploaded=0,failed=0;
  try{
    for(const f of [...input.files]){
      const type=f.type.startsWith('image/')?'image':f.type.startsWith('video/')?'video':f.type.startsWith('audio/')?'audio':'';
      if(!type){failed++;toast(f.name+' অনুমোদিত নয়।');continue}
      const max=type==='image'?20:50;
      if(f.size>max*1024*1024){failed++;toast(f.name+' — সর্বোচ্চ '+max+' MB।');continue}
      toast('আপলোড হচ্ছে: '+f.name+' — 0%');
      const dataUrl=await new Promise(function(resolve,reject){
        const fr=new FileReader();
        fr.onload=function(){resolve(fr.result||'');};
        fr.onerror=function(){reject(new Error('ফাইল পড়া যায়নি।'));};
        fr.readAsDataURL(f);
      });
      const r=await call('saveMedia',token,{name:f.name,type:type,size:f.size,mimeType:f.type,dataUrl:dataUrl,sort:0});
      if(r&&r.ok){uploaded++;toast('আপলোড 100% সম্পন্ন: '+f.name);}
      else{failed++;toast((r&&r.message)||('আপলোড ব্যর্থ: '+f.name));}
    }
    media=await call('listMedia',token);media=Array.isArray(media)?media:[];
    mediaIndex=media.length?Math.min(mediaIndex,media.length-1):0;
    renderMedia();if(typeof renderMediaList==='function')renderMediaList();
    if(uploaded&&failed)toast(uploaded+'টি আপলোড হয়েছে, '+failed+'টি ব্যর্থ।');
    else if(uploaded)toast(uploaded+'টি মিডিয়া সফলভাবে আপলোড হয়েছে।');
    else toast('কোনো মিডিয়া আপলোড হয়নি।');
    input.value='';
  }catch(e){toast('আপলোডে সার্ভার ত্রুটি: '+(e&&e.message?e.message:String(e)))}
}

function dragWindow(el){
  const head=el.querySelector('.win-head');
  if(!head)return;
  head.onmousedown=function(e){
    if(e.target.closest('button'))return;
    const r=el.getBoundingClientRect(),sx=e.clientX,sy=e.clientY;
    el.style.transform='none';
    el.style.left=r.left+'px';el.style.top=r.top+'px';
    function mv(ev){
      let x=r.left+ev.clientX-sx, y=r.top+ev.clientY-sy;
      const maxX=Math.max(0,window.innerWidth-el.offsetWidth), maxY=Math.max(0,window.innerHeight-8);
      x=Math.max(0,Math.min(maxX,x)); y=Math.max(0,Math.min(maxY,y));
      el.style.left=x+'px';el.style.top=y+'px';
    }
    function up(){document.removeEventListener('mousemove',mv);document.removeEventListener('mouseup',up);}
    document.addEventListener('mousemove',mv);document.addEventListener('mouseup',up);
  };
}
